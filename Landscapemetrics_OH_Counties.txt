library(landscapemetrics)

nlcd_raster <- rast("../Spatial/Annual_NLCD_LndCov_2016_CU_C1V0.tif")

us_counties <- st_read("../data/raw/US Counties/tl_2024_us_county.shp")

# only keeping OH counties
ohio_counties <- us_counties %>% filter(STATEFP == "39")

#transforming to match CRS
ohio_counties <- st_transform(ohio_counties, crs(nlcd_raster))

# converting to spat vector
ohio_vect <- vect(ohio_counties)

# Cropping the NLCD raster to keep only OH
nlcd_crop <- crop(nlcd_raster, ohio_vect)
nlcd_mask <- mask(nlcd_crop, ohio_vect)

# Checking
plot(nlcd_mask)

# Computing lsm_c_pland (proportion of NLCD landcover class coverage)
pland<- pblapply(1:nrow(ohio_counties), function(i) {
  tryCatch({
    plands <- sample_lsm(
      landscape = nlcd_mask,
      y = ohio_vect[i],
      what = "lsm_c_pland"  
    )
    plands$id <- i
    return(plands)
  }, error = function(e) {
    message(sprintf("Error at point %d: %s", i, e$message))
    NULL
  })
})


# Combining results 
pland_df <- do.call(rbind, pland)

# wide format
pland_wide <- pland_df %>%
  dplyr::select(id, class, value) %>%
  pivot_wider(
    names_from = class,
    values_from = value,
    names_prefix = "class_"
  )

# Joining 
ohio_counties <- bind_cols(ohio_counties, pland_wide[match(1:nrow(ohio_counties), pland_wide$id), ])


# visualizing and checking a county
Ohio_hancock_county <- ohio_counties %>% filter(COUNTYFP == "063")

Ohio_hancock_county <- st_transform(Ohio_hancock_county, crs(nlcd_raster))


crop<-crop(nlcd_raster, vect(Ohio_hancock_county))
mask<-mask(crop,  vect(Ohio_hancock_county))

# Convert raster to data frame
nlcd_df <- as.data.frame(mask, xy = TRUE)



# Renaming the landcover column
names(nlcd_df)[3] <- "landcover"

# Plot
plot1= ggplot() +
  geom_raster(data = nlcd_df, aes(x = x, y = y, fill = factor(landcover))) +
  geom_sf(data = Ohio_hancock_county, fill = NA, color = "white", size = 1) +
  coord_sf()+
  labs(title = "NLCD Landcover within Hancock County", fill = "Landcover Class") +
  theme_minimal()
plot1
